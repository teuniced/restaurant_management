class ManagementMain
    initialize()
        Create Scanner, Menu, CustomerManager, OrderManager, RecipeBook, Authentication

    run()
        Show welcome message
        while (user doesn't choose to exit)
            If no user logged in
                Attempt login
            Display main menu
            Get user choice
            Handle user choice

    displayMainMenu()
        Show options for different system features

    getUserChoice()
        Get and validate user input

    manageUserChoice(choice)
        Switch based on user's choice
            Case 1: Manage Menu
            Case 2: Process Orders
            Case 3: Manage Customers
            Case 4: Access Recipe Book (with permission check)
            Case 5: View All Data
            Case 6: Logout
            Case 7: Exit

    viewAllData()
        Show all customers and orders

    main()
        ManagementMain instance
        Call run() method

class Authentication
    login(scanner)

    getCurrentUser()
        Return current logged-in user

    logout()
        Clear current user

class CustomerManager
    manageCustomers(scanner)
        Show customer management menu
        Handle user choice (add, view, return)

    addNewCustomer(scanner)
        Get customer details
        Create and save new customer

    viewAllCustomers()
        Show all customers

    loadCustomersFromFile()
        Read customer data from file

    saveCustomerToFile(customer)
        Write customer data to file

class OrderManager
    manageOrders(scanner)
        Show order management menu
        Handle user choice (add, view, return)

    enterANewOrder(scanner)
        Get order details
        Create and save new order

    viewAllOrders()
        Show all orders

    loadOrdersFromFile()
        Read order data from file

    saveOrderToFile(order)
        Write order data to file

class RecipeBook
    loadRecipes()
        Read recipes from file

    viewRecipes(user)
        If the user has access
            Show all recipes
        Else
            Show access denied message

main()
    Create ManagementMain instance
    Call run() method
